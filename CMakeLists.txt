cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(MoonlightUnlockCP CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_definitions(-DUNICODE -D_UNICODE -DWIN32_LEAN_AND_MEAN)
add_library(MoonlightUnlockCP SHARED MoonlightUnlockCP.cpp MoonlightUnlockCP.def)
target_link_libraries(MoonlightUnlockCP ole32 shlwapi secur32 crypt32 advapi32 user32)
